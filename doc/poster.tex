% (1) Project overview  [portrait: col A, row 1]
\postercard{52}{640}{408}{880}{cGreen}{1}{What we do \& why}
\begin{scope}[shift={(20,320)}]
\node[anchor=north west, text width=320mm, align=left, text=cInk,
      font=\fontsize{23}{31}\selectfont] at (50,524)
  {\bull{cGreen} \sub{The problem.}\, Language models read sentences well but lose the
   document as a \emph{whole}; in a legal judgment the decisive signal is spread across
   many paragraphs, and expert labels are scarce (\textbf{few-shot}).\\[15pt]
   \bull{cGreen} \sub{What we build.}\, A frozen \textbf{teacher} reads the clean
   judgment; a \textbf{student} reads a masked, partial copy and must reproduce the
   teacher's reading at \textbf{three scales} --- paragraph, partial- and
   whole-document --- squeezed through a compressive \textbf{information
   bottleneck}.\\[15pt]
   \bull{cGreen} \sub{The roots.}\, It adapts the \emph{Glocal Information Bottleneck}
   (Yang 2025) from time series to legal text, and is tested against hierarchical
   MLM (\textbf{H-MLM}) and a no-pretrain baseline on \textbf{ECtHR} judgments
   (LexGLUE; 10 labels, ${\approx}100\times$ imbalance, macro-F1).\\[15pt]
   \bull{cGreen} \sub{We ask three questions.}\\[7pt]
   \hspace*{7mm}\textbf{RQ1}~ Does GlocalDoc beat a hierarchical MLM at few-shot?\\[6pt]
   \hspace*{7mm}\textbf{RQ2}~ Does the advantage depend on the label budget \emph{N}?\\[6pt]
   \hspace*{7mm}\textbf{RQ3}~ What breaks on boilerplate, and what fixes it?};

\end{scope}

% (3) Preventing collapse  [portrait: col A, row 2]
\postercard{52}{430}{408}{624}{cTeal}{3}{Preventing collapse}
\begin{scope}[shift={(20,322)}]
\node[anchor=north west, text width=320mm, align=left, text=cInk,
      font=\fontsize{24}{37}\selectfont] at (50,266)
  {\sub{The failure.} ECtHR judgments are mostly boilerplate; the raw encoder sees
   documents as \emph{nearly identical} (median pairwise cosine \textbf{0.9997}). The
   teacher target becomes constant and the student \textbf{collapses}.\\[20pt]
   \sub{The fix.} \textbf{DINO centring} subtracts a running mean from the teacher
   target. Inter-document cosine of the student code falls \textbf{0.42 to
   \textless0.05} and stays stable: the single change that keeps training stable.\\[20pt]
   \sub{Also.} EMA teacher ($\tau$\,=\,0.996), predictor on the student only,
   VICReg variance \& covariance.};

\end{scope}
